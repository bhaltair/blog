    2. 使用https

    var options = {
      key: fs.readFileSync(__dirname + 'ca.pem'),
      cert: fs.readFileSync(__dirname + 'ca.crt')
    }
    https.createServer(options, app).listen(app.get('port'), function(){
      console.log('app started at ' + app.get('port'))
    })
