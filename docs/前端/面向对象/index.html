    属性类型

    数据属性

    • value：就是属性的值，可以为函数
    • writable：决定属性能否被赋值
    • enumerable：决定 for in 能否枚举该属性
    • configurable：决定该属性能否被删除或者改变特征值
    var person = {}
    Object.defineProperty(person, 'name', {
      configurable: false,
      value: 'jason',
      writable: false,
      enumberable: false
    })

    访问器属性

    • [[Configurable]] 决定该属性能否被配置
    • [[Enumberable]] 决定 for in 能否枚举该属性
    • [[Get]] 在取属性值时被调用
    • [[Set]] 在设置属性值时被调用
    var person = {}
    Object.defineProperty(person, 'name', {
      configurable: false,
      enumerable: false,
      get: function(value) {
      },
      set: function(value) {
      }
    })

    访问原型

    • Object.create 根据指定的原型创建新对象，原型可以是 null，可以接收第二个参数
    • Object.getPrototypeOf 获得对象的原型
    • Object.setPrototypeOf 设置对象的原型

    创建对象

    1. 工厂模式
    function createPerson(name, age, job){
          var o = new Object();
          o.name = name;
          o.age = age;
          o.job = job;
          o.sayName = function(){
            alert(this.name)
          }
          retrun o;
    var person1 = createPerson('Nico', 29, 'Software Engineer')
    }
    1. 构造函数
    function Person(name, age, job){
          this.name = name;
          this.age = age;
          this.job = job;
          this.sayName = function() {
            alert(this.name)
          }
    }

    优点：

    没有显示的创建对象
    没有return语句

    缺点：1. 每个方法都要在实例上创建一遍

    1. 原型模式

      所有实例共享原型对象的属性和方法

      * 原型对象上存在constructor属性，指向它的构造函数
      * 构造函数上有一个prototype属性，指向它的原型对象
      * 实例Instance上有一个__proto__指针，指向它的prototype对象
      * 实例上的同名属性，并不会覆盖原型对象上的值，只不过查找值，先从实例上找

    2. 寄生构造函数模式

      和工厂模式很类似,只不过是用New操作符 

    function Person(name, age, job){
          var o = new Object();
          o.name = name;
          o.age = age;
          o.job = job;
          o.sayName = function(){
            alert(this.name)
          }
          retrun o;
    var person1 = new Person('Nico', 29, 'Software Engineer')

    返回的对象与构造函数或者与构造函数的原型属性之间没有关系

    理解原型链的查找

    先在实例对象上找，如果找不到，那么去原型对象上找，如果原型对象上找不到，那么原型对象的原型上找

    继承

    原型式继承

    SubType.prototype = new SuperType()

    问题：父类实例上的属性变为现在的原型属性了。

    ### 组合继承
    *最常用的继承模式*
    组合继承 = 借用构造函数 + 原型继承
    > 使用原型链继承共享的属性和方法，而通过借用构造函数继承实例属性。
    ```javascript
    function Super(name) {
      this.name = name;
      this.colors = ['red', 'blue'];
    }
    Super.prototype.sayName = function() {
      alert(this.name)
    }
    function SubType(name, age){
      Super.call(this, name);
      this.age = age;
    }
    SubType.prototype = new SuperType();
    var instance = new SubType()
    console.log(instance instanceof SubType) //true
    console.log(instance instanceof Super) //true

    缺点：调用两次父类构造函数

    寄生组合式继承

    组合继承和原型式继承的结合 最理想的继承范式。

    1. 创建超类型原型的一个副本 利用Object.creat()
    2. 为创建的副本添加constructor属性，指向子类构造函数
    3. 新创建的对象副本复制给子类型的原型
    function Super(name) {
      this.name = name;
      this.colors = ['red', 'blue'];
    }
    Super.prototype.sayName = function() {
      alert(this.name)
    }
    //SubType.prototype = new SuperType();  不用这句了
    function inherit(o){
      function F(){}
      F.prototype = o;
      return new F();
    }
    function inheritPrototype(sub,Super){
      
      //下面2句都是可以 创建父类原型的一个副本
      //var proto = inherit(Super.prototype);
      var proto = Object.create(super.prototype)
      proto.constructor = sub;  //重写原型
      sub.prototype = proto;
    }
    inheritPrototype(SubType, Super);
    function SubType(name, age){
      Super.call(this, name);  //
      this.age = age;
    }
    SubType.prototype.sayAge = function() {
      alert(this.age);
    }

    优点： 只调用一次父类构造函数，避免了在SubType.prototype上面创建不必要的属性；而且还能正常用instanceof和isPrototypeOf()

    原生插件

    ;(function(doc,win){
    var Plugins = function(options) {  
       this.setting = {
            pluginsID : '',
            alertHtml : ''
        };
        for(var i in this.setting){
            this.setting[i] = options[i] || this.setting[i];
        }
        // 使用必要条件进行检测，然后再往其原型中添加函数。  
        if(typeof this.NULL != "function") {  
            // 缓存Plugins.prototype
            var proto = Plugins.prototype;  
            // 抽象获取元素id方法
            proto.g = function(id){
                return document.getElementById(id);
            },
            // 抽象元素绑定事件
            proto.on = function(id,type,fn){
                this.g(id)['on'+type] = fn;
            },
            proto.init = function(){
                var self = this;
                self.on(self.setting.pluginsID,'click',function(){
                    alert(self.setting.alertHtml)
                })
            }
        }  
    }; 
    window['Plugins'] = Plugins;
    })();
    new Plugins({
        pluginsID : 'plugins',
        alertHtml : '欢迎来到插件世界'
    }).init();
